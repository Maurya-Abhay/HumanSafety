// Police submits registration request (CANNOT self-activate)

Body: {

  name, 
  stationName, 
  badgeNumber, 
  idProof, 
  badgeProof 
}
Status: pending (awaiting admin approval)

GET /police/officers
Auth: Required

// ============== ADMIN ONLY ==============

Auth: Required + Admin role

Body: { adminNotes }

// Admin rejects police request
POST /police/reject/:userId
Auth: Required + Admin role
Body: { rejectionReason }

// ============================================================
// AUTHENTICATION MIDDLEWARE
// ============================================================

// Role-based authorization
requireRole('admin', 'police', 'hospital', 'user')

// Approval status check (rejects pending/rejected/blocked)
requireApproved

// Block status check
requireNotBlocked

// ============================================================
// ROLE SYSTEM
// ============================================================

User Roles:
- 'user': Regular citizen (can use panic button, request help)
- 'police': Police officer (can accept cases, navigate, provide help)
- 'hospital': Hospital (can accept emergencies, manage beds)
- 'admin': Administrator (approves, blocks, views analytics)

User Status:
- 'active': Account approved and operational
- 'pending': Awaiting admin approval (Police/Hospital only)
- 'rejected': Registration request rejected
- 'blocked': Account disabled by admin

STRICT LOGIC:
- User: Can register anytime
- Police: CANNOT self-activate, requires admin approval via /police/approve/:userId
- Hospital: CANNOT self-activate, requires admin approval via /hospital-admin/approve/:hospitalId
- Admin: Has full control over all accounts and approvals

// ============================================================
// WORKFLOW: Emergency Response
// ============================================================

1. User presses panic button
   POST /emergency/panic -> creates Alert with status='pending'

2. Dispatcher assigns case to police
   POST /case/assign -> Alert.status = 'assigned', Alert.assignedPolice = policeId

3. Police accepts or rejects
   POST /case/:caseId/accept -> status = 'in-progress'
   POST /case/:caseId/reject -> status = 'pending' (back to unassigned)

4. Police updates status while responding
   PUT /case/:caseId/status -> status = 'at-location', 'providing-help'

5. Case resolved
   POST /case/:caseId/resolve -> status = 'resolved', resolvedAt = now
